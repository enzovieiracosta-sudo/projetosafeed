Código do Filtro Anti-Ódio (SDK)

Lógica de bloqueio estrito pronta para integração.

// Função de filtro com tolerância zero para racismo e ofensas
function filtrarComentario(texto) {
  const termosProibidos = ["macaco", "preto feio", "lixo", "racista", "odeio essa raça", "volta pra sua terra"];
  const contemOfensa = termosProibidos.some(termo => texto.toLowerCase().includes(termo));

  if (contemOfensa) {
    // Apaga o comentário e impede a exibição
    return { exibido: false, motivo: "Conteúdo ofensivo ou racista apagado pelo SAFEED" };
  }
  return { exibido: true, conteudo: texto };
}
